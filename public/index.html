const form = document.getElementById("submissionForm"); const list = document.getElementById("submissionList"); // Function to render submissions on UI function renderSubmissions(submissions) { list.innerHTML = ""; submissions.forEach((item) => { const li = document.createElement("li"); li.innerHTML = `${item.name} (${item.regNo}) - ${item.title}: ${item.url}`; list.appendChild(li); }); } // Load initial submissions (from server or localStorage) async function loadSubmissions() { try { const res = await fetch("/api/submissions"); if (res.ok) { const data = await res.json(); renderSubmissions(data); return; } } catch (err) { // Fallback to localStorage if static/GitHub Pages } const localData = JSON.parse(localStorage.getItem("submissions") || "[]"); renderSubmissions(localData); } // Handle form submission form.addEventListener("submit", async (e) => { e.preventDefault(); const payload = { name: document.getElementById("name").value, regNo: document.getElementById("regNo").value, title: document.getElementById("title").value, url: document.getElementById("url").value, }; try { const res = await fetch("/api/submissions", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload), }); if (res.ok) { form.reset(); loadSubmissions(); return; } } catch (err) { // Static environment / GitHub Pages fallback } // Client-side fallback for static GitHub Pages const localData = JSON.parse(localStorage.getItem("submissions") || "[]"); localData.push(payload); localStorage.setItem("submissions", JSON.stringify(localData)); form.reset(); renderSubmissions(localData); alert("Submission saved locally!"); }); loadSubmissions();
